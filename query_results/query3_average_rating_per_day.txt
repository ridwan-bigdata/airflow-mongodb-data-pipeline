QUERY 3: Average Rating for Each Day

MongoDB Query:

db.reviews.aggregate([
  {
    $group: {
      _id: {
        $dateTrunc: {
          date: {
            $toDate: "$at"
          },
          unit: "day"
        }
      },
      avg_rating: {
        $avg: "$score"
      }
    }
  },
  {
    $sort: {
      _id: 1
    }
  }
])
